cmake_minimum_required(VERSION 3.16)
project(nd_runtime C CXX)
set(CMAKE_C_STANDARD 11)
find_package(Threads REQUIRED)
add_library(nd_runtime STATIC client.c device.c session.c)
set_target_properties(nd_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(nd_runtime PUBLIC ../include)
target_link_libraries(nd_runtime PUBLIC Threads::Threads)
add_library(nd_runtime_shared SHARED client.c device.c session.c)
set_target_properties(nd_runtime_shared PROPERTIES OUTPUT_NAME nd_runtime)
target_include_directories(nd_runtime_shared PUBLIC ../include)
target_link_libraries(nd_runtime_shared PRIVATE Threads::Threads)
option(ND_BUILD_EXECUTOR "Build the libffi device executor" ON)
find_path(FFI_INCLUDE_DIR ffi.h PATH_SUFFIXES x86_64-linux-gnu arm-linux-gnueabihf)
find_library(FFI_LIBRARY ffi)
if(ND_BUILD_EXECUTOR)
 if(NOT FFI_INCLUDE_DIR OR NOT FFI_LIBRARY)
  message(FATAL_ERROR "Executor needs libffi headers/library. Set FFI_INCLUDE_DIR and FFI_LIBRARY, or ND_BUILD_EXECUTOR=OFF for client only.")
 endif()
 add_executable(nd-executor executor.c)
 target_include_directories(nd-executor PRIVATE ${FFI_INCLUDE_DIR} ../include)
 target_link_libraries(nd-executor PRIVATE ${FFI_LIBRARY} ${CMAKE_DL_LIBS})
endif()

include(GNUInstallDirs)
install(TARGETS nd_runtime nd_runtime_shared
 ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ../include/nd_runtime.h ../include/nd_device.h
 DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

add_library(nd SHARED ../src/libnd.cpp)
add_library(nd_static STATIC ../src/libnd.cpp)
set_target_properties(nd_static PROPERTIES OUTPUT_NAME nd)
foreach(target nd nd_static)
 target_include_directories(${target} PUBLIC ../include)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_link_libraries(${target} PUBLIC nd_runtime_shared)
endforeach()
install(TARGETS nd nd_static ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ../include/nd.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
